      136 +// short-piece machinery (supervisor test): fit fixed windows 
          +of L consecutive                                              
      137 +// layers independently with the SAME estimator, pool residual
          +s per track.                                                  
      138 +// sqrt(sum chi2 / sum (n-3)) is dof-unbiased: a 4-point piece
          + has 1 dof, so                                                
      139 +// raw sqrt(chi2/n) would read sigma/2 and deflate the sparse 
          +(sim) side.                                                   
      140 +struct Seg { double rms = 0; int npiece = 0; long npts = 0; };
      141 +bool segFit(const GrpL &G, int L, Seg &S)                     
      142 +{                                                             
      143 +  std::map<int, std::vector<int>> win;              // (lay-7)
          +/L -> point idx                                               
      144 +  for (size_t i = 0; i < G.lay.size(); ++i) win[(G.lay[i] - 7)
          + / L].push_back((int) i);                                     
      145 +  double chi2 = 0;                                            
      146 +  long dof = 0;                                               
      147 +  for (auto &kv : win)                                        
      148 +  {                                                           
      149 +    if ((int) kv.second.size() < 4) continue;                 
      150 +    std::set<int> rows;                                       
      151 +    for (int i : kv.second) rows.insert(G.lay[i]);            
      152 +    if ((int) rows.size() < 3) continue;            // need an
          + arc, not a pad column                                        
      153 +    std::vector<double> X, Y;                                 
      154 +    for (int i : kv.second) { X.push_back(G.x[i]); Y.push_back
          +(G.y[i]); }                                                   
      155 +    Fit F = fitCircle(X, Y, 4);                               
      156 +    if (!F.ok || !std::isfinite(F.rms)) continue;             
      157 +    chi2 += F.rms * F.rms * F.n;                              
      158 +    dof += F.n - 3;                                           
      159 +    S.npiece++;                                               
      160 +    S.npts += F.n;                                            
      161 +  }                                                           
      162 +  if (S.npiece < 1 || dof < 1) return false;
